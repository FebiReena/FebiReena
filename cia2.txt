Sure. Below is the **complete simple Angular NgModule program** with:

* ✅ Module
* ✅ Components
* ✅ Routing that actually changes pages
* ✅ Forms
* ✅ CRUD operations
* ✅ Student table
* ✅ Edit / Delete
* ✅ **Moving advertisement image from left → right**
* ✅ Simple CSS
* ✅ Angular animation for the advertisement

# 8. Student Component

This is where **Forms + CRUD + Angular Animation** are demonstrated.

## `student.component.ts`

**`src/app/student/student.component.ts`**

```ts
import { Component } from '@angular/core';

import {
  trigger,
  transition,
  style,
  animate
} from '@angular/animations';

@Component({
  selector: 'app-student',
  templateUrl: './student.component.html',
  styleUrls: ['./student.component.css'],

  animations: [

    trigger('fade', [

      transition(':enter', [

        style({
          opacity: 0
        }),

        animate(
          '500ms',
          style({
            opacity: 1
          })
        )

      ])

    ])

  ]
})
export class StudentComponent {

  id = '';

  name = '';

  department = '';

  gender = '';

  hobbies: string[] = [];

  students: any[] = [];

  editIndex = -1;


  // CREATE / UPDATE

  addStudent() {

    const student = {

      id: this.id,

      name: this.name,

      department: this.department,

      gender: this.gender,

      hobbies: this.hobbies.join(', ')

    };


    if (this.editIndex == -1) {

      // CREATE

      this.students.push(student);

    }

    else {

      // UPDATE

      this.students[this.editIndex] = student;

      this.editIndex = -1;

    }


    this.clearForm();

  }


  // UPDATE

  editStudent(index: number) {

    this.editIndex = index;

    this.id = this.students[index].id;

    this.name = this.students[index].name;

    this.department =
      this.students[index].department;

    this.gender =
      this.students[index].gender;

    this.hobbies =
      this.students[index].hobbies.split(', ');

  }


  // DELETE

  deleteStudent(index: number) {

    this.students.splice(index, 1);

  }


  // CHECKBOX

  selectHobby(hobby: string) {

    if (this.hobbies.includes(hobby)) {

      this.hobbies =
        this.hobbies.filter(h => h !== hobby);

    }

    else {

      this.hobbies.push(hobby);

    }

  }


  // CLEAR FORM

  clearForm() {

    this.id = '';

    this.name = '';

    this.department = '';

    this.gender = '';

    this.hobbies = [];

    this.editIndex = -1;

  }

}
```

## What each part demonstrates

| Requirement    | Where                          |
| -------------- | ------------------------------ |
| **Module**     | `app.module.ts`                |
| **Component**  | Home, Student, About           |
| **Routing**    | `app-routing.module.ts`        |
| **Navigation** | `routerLink`                   |
| **Forms**      | `[(ngModel)]`                  |
| **Create**     | `students.push()`              |
| **Read**       | `*ngFor`                       |
| **Update**     | `editStudent()`                |
| **Delete**     | `splice()`                     |
| **Animation**  | Moving advertisement + `@fade` |
| **Table**      | `student.component.html`       |

**One important point:** the moving advertisement is a **CSS animation using `@keyframes`**, while the table-row fade is an **Angular animation using `@angular/animations`**. So you actually have both types in this program.
